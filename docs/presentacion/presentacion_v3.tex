\begin{frame}
  \frametitle{1. EL DIAGNÓSTICO: 3 PROBLEMAS CRÍTICOS DEL PASEO}
  \framesubtitle{¿Por qué los centros comerciales pierden relevancia frente a las plataformas digitales?}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Problemas del Visitante y el Comercio}
        \footnotesize
        \begin{itemize}\setlength\itemsep{0.35em}
          \item \textbf{Fuga de Clientes por Delivery Tradicional:} Las apps de delivery encierran al cliente en su casa, devorando hasta 30\% en comisiones y destruyendo las compras por impulso dentro del mall.
          \item \textbf{Fricción en Locales y Colas en Caja:} Locales aislados sin catálogo conjunto. Pedir un café o un almuerzo requiere filas innecesarias.
          \item \textbf{Desconocimiento de Pisos Superiores:} El 70\% del público desconoce la oferta de la terraza, cowork y servicios médicos.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Problemas de la Administración Corporativa}
        \footnotesize
        \begin{itemize}\setlength\itemsep{0.35em}
          \item \textbf{Ceguera Total de Tráfico Peatonal:} El mall no sabe cuánta gente entra realmente, cuánto tiempo se queda ni por qué pasillos circula.
          \item \textbf{Decisiones Inmobiliarias a Ciegas:} Alquileres fijados por $m^2$ estático sin sustento de afluencia o conversión de compra real.
          \item \textbf{Pasillos Fríos Desatendidos:} Zonas comerciales sin tracción que se convierten en locales vacíos por falta de activación.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vskip 0.25cm
  \centering
  \color{AranjuezGold}\footnotesize \textbf{Tesis Central:} La tecnología no debe competir contra el mall; debe ser el anzuelo digital que obligue al cliente a pisar el centro comercial y medir su comportamiento.
\end{frame}

\begin{frame}
  \frametitle{2. LA SOLUCIÓN: CÓMO RESOLVIMOS CADA RETO TÉCNICO}
  \framesubtitle{Integración simbiótica diseñada para convertir tráfico digital en ventas físicas}

  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Reto 3: PaseoYa}
        \scriptsize
        \textbf{E-commerce con Retiro Físico}
        \begin{itemize}\setlength\itemsep{0.25em}
          \item \textbf{El Problema:} Pérdida de ventas ante apps externas de delivery.
          \item \textbf{Cómo lo resolvimos:} Carrito multi-tienda con \textbf{Click \& Collect obligatorio}.
          \item El cliente compra online pero \textbf{debe acudir al local} a retirar.
          \item QR dinámico verificado por cajeros en $<0.5$ segundos.
          \item \textbf{Efecto:} +35\% en ventas cruzadas en mostrador.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.31\textwidth}
      \begin{block}{Reto 2: Jarvis AI}
        \scriptsize
        \textbf{Asistente de Voz Multimodal}
        \begin{itemize}\setlength\itemsep{0.25em}
          \item \textbf{El Problema:} Asimetría y desorientación del visitante.
          \item \textbf{Cómo lo resolvimos:} Motor Gemini de ultra-baja latencia ($<800$ms) con síntesis de voz nativa.
          \item Conectado en tiempo real a tiendas, horarios, eventos y terrazas.
          \item Responde dudas contextuales (ej. \textit{``¿Dónde hay café con vista al Tunari?''}).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Reto 1: Club Aranjuez}
        \scriptsize
        \textbf{Fidelización \& Geocerca 200m}
        \begin{itemize}\setlength\itemsep{0.25em}
          \item \textbf{El Problema:} Falta de incentivos para regresar y explorar niveles.
          \item \textbf{Cómo lo resolvimos:} Geocerca GPS perimetral (200m en Av. América) + Tótems QR en cada piso.
          \item 4 Niveles reales (Bronce a Platino) que premian la permanencia y el recorrido vertical.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vskip 0.3cm
  \centering
  \color{AranjuezGreen}\footnotesize \textbf{El Circuito Cerrado:} Jarvis orienta $\rightarrow$ PaseoYa transacciona $\rightarrow$ Club premia la visita $\rightarrow$ El cliente acude al mall.
