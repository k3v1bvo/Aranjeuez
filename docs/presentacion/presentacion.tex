\begin{frame}
  \frametitle{1. EL DIAGNÓSTICO: LA PARADOJA DEL RETAIL FÍSICO}
  \framesubtitle{El comercio electrónico tradicional extrae a los consumidores de los centros comerciales}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{La Amenaza: Desconexión del Tráfico}
        \small
        \begin{itemize}\setlength\itemsep{0.35em}
          \item \textbf{Canibalización por Delivery:} Las plataformas convencionales encierran al cliente en casa, extinguiendo el consumo colateral del centro comercial.
          \item \textbf{Asimetría de Información:} Los clientes desconocen el 70\% de la oferta comercial y médica de los pisos superiores.
          \item \textbf{Fidelización Ineficiente:} Programas analógicos desconectados que no generan incentivos para regresar.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{La Brecha Operativa: Ceguera de Datos}
        \small
        \begin{itemize}\setlength\itemsep{0.35em}
          \item \textbf{Sin Métricas de Permanencia:} La administración carece de datos sobre el tiempo de estadía real (\textit{dwell time}) por nivel o pasillo.
          \item \textbf{Rentas Fijadas a Ciegas:} No existe sustento analítico cuantitativo para valorar locales comerciales de alto vs. bajo flujo.
          \item \textbf{Comercios Aislados:} Falta de sinergia cruzada entre gastronomía, servicios y retail corporativo.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vskip 0.3cm
  \centering
  \color{EmeraldLight}\footnotesize \textbf{Tesis Estratégica:} La tecnología no debe reemplazar la visita al mall, sino convertirse en el catalizador que la provoque y la mida.
\end{frame}
